# cppcheck suppressions for the C kernel gate (docs/CODEBASE-INTEL.md).
#
# The gate runs with --enable=warning,performance,portability and
# --error-exitcode=1, so every entry here is a documented exception, not a
# way to hide a class of findings. Style findings are not suppressed at all:
# they are ratcheted by scripts/codebase_intel.py instead.

# odk_core.c: ODK_LIBRARY_VERSION is provided by the build system (Makefile /
# CMakeLists pass it via -D); cppcheck cannot see it without a compile
# database, so the "unknown macro" report is a configuration artifact.
unknownMacro:vendor/omnidecide-c/src/odk_core.c

# odk_learn.c:1627 - `sep - p` is computed before the `sep == NULL` guard on
# the error path that discards the value. Benign in practice (the value is
# never read when sep is NULL) but technically pointer arithmetic on NULL.
# Pre-existing in the vendored, golden-corpus-frozen kernel; tracked for an
# upstream reorder, not fixed here so the frozen kernel stays byte-identical.
nullPointerArithmeticRedundantCheck:vendor/omnidecide-c/src/odk_learn.c:1627
